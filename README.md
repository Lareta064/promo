# Инструкция по запуску и сборке проекта

npm install      # установить зависимости
npm run dev      # запустить проект для разработки
npm run build    # собрать финальную версию в папку build
npm run preview  # посмотреть собранную версию

# Что передавать заказчику

build/           # готовая собранная версия проекта
node_modules/    # не передавать, создаётся через npm install
